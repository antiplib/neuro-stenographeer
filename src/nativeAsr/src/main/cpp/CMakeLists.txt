cmake_minimum_required(VERSION 3.22)
project(steno_asr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/whisper ${CMAKE_BINARY_DIR}/whisper)
add_library(steno_asr SHARED bridge.cpp)
target_link_libraries(steno_asr PRIVATE whisper log)
# Each engine has its own static ggml. Do not expose or interpose its symbols.
target_link_options(steno_asr PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-z,max-page-size=16384")

set(SHERPA_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/sherpa)
add_library(sherpa_c_api SHARED IMPORTED)
set_target_properties(sherpa_c_api PROPERTIES IMPORTED_LOCATION ${SHERPA_ROOT}/jniLibs/${ANDROID_ABI}/libsherpa-onnx-c-api.so)
target_include_directories(steno_asr PRIVATE ${SHERPA_ROOT}/include)
target_link_libraries(steno_asr PRIVATE sherpa_c_api)
